MERGE `zapier-data-471820.cloud_function.current_table` AS T
USING (
  SELECT AS STRUCT *
  FROM (
    SELECT *,
      ROW_NUMBER() OVER (
        PARTITION BY Deal_ID
        ORDER BY COALESCE(Updated, TIMESTAMP '1970-01-01') DESC
      ) AS rn
    FROM `zapier-data-471820.cloud_function.raw_table`
    WHERE Updated >= TIMESTAMP_SUB(CURRENT_TIMESTAMP(), INTERVAL 25 HOUR)
      -- AND Deal_ID = 1464  -- ✅ Limit to one deal for testing
  )
  WHERE rn = 1
) AS S
ON T.Deal_ID = S.Deal_ID
WHEN MATCHED AND COALESCE(S.Updated, TIMESTAMP '1970-01-01') > COALESCE(T.Updated, TIMESTAMP '1970-01-01')
THEN
  UPDATE SET
    Title = S.Title,
    Description = S.Description,
    Value = S.Value,
    Status = S.Status,
    Owner_Name = S.Owner_Name,
    Pipeline = S.Pipeline,
    Stage = S.Stage,
    Account = S.Account,
    Created = S.Created,
    Updated = S.Updated,
    Next_Action_Date = S.Next_Action_Date,
    Forecasted_Close_Date = S.Forecasted_Close_Date,
    Estimated_Equipment = S.Estimated_Equipment,
    Project_Completed = S.Project_Completed,
    Description_of_Work = S.Description_of_Work,
    Business_Classification = S.Business_Classification,
    Closing_Date_Probability = S.Closing_Date_Probability,
    Project_Address = S.Project_Address,
    RMR_Bid_Amount = S.RMR_Bid_Amount,
    RMR_Quoted_Term_in_Months = S.RMR_Quoted_Term_in_Months,
    RMR_Quoted_Payment_Terms = S.RMR_Quoted_Payment_Terms,
    Type = S.Type,
    Sales_Person = S.Sales_Person,
    Type_of_sale = S.Type_of_sale,
    Source = S.Source,
    Deal_Lost_Reason = S.Deal_Lost_Reason,
    Time_In = S.Time_In,
    Time_Out = S.Time_Out,
    Sentiment_to_close = S.Sentiment_to_close,
    GM_percent = S.GM_percent,
    GM_dollar = S.GM_dollar,
    System_Type = S.System_Type,
    Calls = S.Calls,
    Texts = S.Texts,
    Emails = S.Emails,
    Proposal_Delivery_Method = S.Proposal_Delivery_Method,
    Deal_Close_Date = S.Deal_Close_Date
WHEN NOT MATCHED THEN
  INSERT (
    Deal_ID,
    Title,
    Description,
    Value,
    Status,
    Owner_Name,
    Pipeline,
    Stage,
    Account,
    Created,
    Updated,
    Next_Action_Date,
    Forecasted_Close_Date,
    Estimated_Equipment,
    Project_Completed,
    Description_of_Work,
    Business_Classification,
    Closing_Date_Probability,
    Project_Address,
    RMR_Bid_Amount,
    RMR_Quoted_Term_in_Months,
    RMR_Quoted_Payment_Terms,
    Type,
    Sales_Person,
    Type_of_sale,
    Source,
    Deal_Lost_Reason,
    Time_In,
    Time_Out,
    Sentiment_to_close,
    GM_percent,
    GM_dollar,
    System_Type,
    Calls,
    Texts,
    Emails,
    Proposal_Delivery_Method,
    Deal_Close_Date
  )
  VALUES (
    S.Deal_ID,
    S.Title,
    S.Description,
    S.Value,
    S.Status,
    S.Owner_Name,
    S.Pipeline,
    S.Stage,
    S.Account,
    S.Created,
    S.Updated,
    S.Next_Action_Date,
    S.Forecasted_Close_Date,
    S.Estimated_Equipment,
    S.Project_Completed,
    S.Description_of_Work,
    S.Business_Classification,
    S.Closing_Date_Probability,
    S.Project_Address,
    S.RMR_Bid_Amount,
    S.RMR_Quoted_Term_in_Months,
    S.RMR_Quoted_Payment_Terms,
    S.Type,
    S.Sales_Person,
    S.Type_of_sale,
    S.Source,
    S.Deal_Lost_Reason,
    S.Time_In,
    S.Time_Out,
    S.Sentiment_to_close,
    S.GM_percent,
    S.GM_dollar,
    S.System_Type,
    S.Calls,
    S.Texts,
    S.Emails,
    S.Proposal_Delivery_Method,
    S.Deal_Close_Date
  );
